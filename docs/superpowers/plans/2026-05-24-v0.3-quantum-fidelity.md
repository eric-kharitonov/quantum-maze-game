```python
def build_nonzero(k):
    """k-qubit equal superposition over all non-zero bit-strings.

    State: |ψ⟩ = (1/√(2^k − 1)) Σ_{x ≠ 0} |x⟩

    Implementation: apply Hadamards on all k qubits to reach the uniform
    superposition over all 2^k strings, then rejection-sample at measurement
    time. We do this in the build step by appending a classical post-filter:
    instead of changing the circuit, the /collapse handler retries
    measurement until a non-zero outcome appears. With k qubits the all-zero
    outcome has probability 1/2^k so even at k=1 retry expectation is 2 shots.
    """
    qc = QuantumCircuit(k, k)
    for i in range(k):
        qc.h(i)
    qc.measure(range(k), range(k))
    return qc
```

```javascript
// v0.3: classical pre-measurement filter. For each SUPERPOSED wall at `cell`:
//   - if neighbor is visited → SOLID (loop prevention)
//   - if neighbor is unvisited but already reachable from {visited} via
//     OPEN walls → SOLID (cycle prevention)
// Returns the array of remaining-SUPERPOSED candidate sides ['N','E','S','W']-style.
function applyLoopAndCyclePrevention(cell) {
  // Step 1: BFS from visited set through OPEN walls → reachable set.
  const reachable = new Set();
  const seedQ = [];
  for (let r = 0; r < GRID; r++) {
    for (let c = 0; c < GRID; c++) {
      if (cells[r][c].visited) {
        reachable.add(`${r},${c}`);
        seedQ.push({ r, c });
      }
    }
  }
  while (seedQ.length) {
    const cur = seedQ.shift();
    for (const side of SIDES) {
      const id = cellWalls(cur.r, cur.c)[side];
      if (!walls[id] || walls[id].state !== 'OPEN') continue;
      const t = sideToCoord(cur, side);
      if (t.r < 0 || t.r >= GRID || t.c < 0 || t.c >= GRID) continue;
      const key = `${t.r},${t.c}`;
      if (reachable.has(key)) continue;
      reachable.add(key);
      seedQ.push(t);
    }
  }

  // Step 2: for each side of `cell`, decide.
  const candidates = [];
  for (const side of SIDES) {
    const id = cellWalls(cell.r, cell.c)[side];
    const w = walls[id];
    if (!w) continue;                       // off-grid edge
    if (w.state !== 'SUPERPOSED') continue; // already decided (open/solid/entangled handled elsewhere)
    if (w.isExit) continue;                 // exit wall never enters the circuit
    if (w.isBorder) {                       // border walls aren't candidates
      // leave as SUPERPOSED; they'll be ignored
      continue;
    }
    const t = sideToCoord(cell, side);
    if (t.r < 0 || t.r >= GRID || t.c < 0 || t.c >= GRID) continue;
    if (cells[t.r][t.c].visited) {
      walls[id].state = 'SOLID';  // loop prevention
      stats.wallsCollapsed++;
      continue;
    }
    if (reachable.has(`${t.r},${t.c}`)) {
      walls[id].state = 'SOLID';  // cycle prevention
      stats.wallsCollapsed++;
      continue;
    }
    candidates.push(side);
  }
  return candidates;
}
```

```javascript
// v0.3: fire one non-zero superposition circuit over `candidates` (an array
// of sides like ['N','E','S']). Mutates wall states accordingly:
//   - each bit_i = 1 → wall_i → OPEN
//   - each bit_i = 0 → wall_i → SOLID
// At least one is guaranteed OPEN (k=0 should never call this).
async function processNonzeroCircuit(cell, candidates) {
  if (candidates.length === 0) return [];

  const candIds = candidates.map(s => cellWalls(cell.r, cell.c)[s]);
  const candCoords = candidates.map(s => {
    const t = sideToCoord(cell, s);
    return [t.r, t.c];
  });
  for (const id of candIds) walls[id].pending = true;
  render();

  let outcomes;
  try {
    outcomes = await Quantum.nonzero([cell.r, cell.c], candCoords);
  } catch (e) {
    console.error('Non-zero circuit call failed', e);
    for (const id of candIds) walls[id].pending = false;
    return [];
  }

  const opened = [];
  for (let i = 0; i < candIds.length; i++) {
    const id = candIds[i];
    walls[id].pending = false;
    walls[id].state = outcomes[i] === 1 ? 'OPEN' : 'SOLID';
    stats.wallsCollapsed++;
    if (outcomes[i] === 1) opened.push(candidates[i]);
  }
  console.log(`[Q] cell (${cell.r},${cell.c}) k=${candidates.length} → outcomes [${outcomes.join(',')}], opened: ${opened.join(',') || '(none — bug)'}`);
  return opened;
}
```

```javascript
// v0.3: after every cell entry, verify the exit cell is still reachable
// from the player's current position via non-SOLID walls. Returns true if
// the maze has been sealed off and the player cannot reach the exit.
function checkExitSealed() {
  if (exitRow === null) return false;
  const target = `${exitRow},${GRID - 1}`;
  const reachable = new Set([`${player.r},${player.c}`]);
  const q = [{ r: player.r, c: player.c }];
  while (q.length) {
    const cur = q.shift();
    if (`${cur.r},${cur.c}` === target) return false;
    for (const side of SIDES) {
      const id = cellWalls(cur.r, cur.c)[side];
      if (!walls[id] || walls[id].state === 'SOLID') continue;
      const t = sideToCoord(cur, side);
      if (t.r < 0 || t.r >= GRID || t.c < 0 || t.c >= GRID) continue;
      const key = `${t.r},${t.c}`;
      if (reachable.has(key)) continue;
      reachable.add(key);
      q.push(t);
    }
  }
  return true;  // exit not in reachable set
}
```

```javascript
async function enterCell(cell) {
  inputLocked = true;
  cells[cell.r][cell.c].visited = true;
  stats.steps++;

  // v0.3 pipeline:
  // 1. Bell measurements on any ENTANGLED walls adjacent to this cell.
  //    (Bell can produce OPEN outcomes that bypass cycle prevention —
  //    these are "Bell wormholes," accepted per the spec.)
  await processBell(cell);

  // 2. Loop prevention + cycle prevention, returning the surviving
  //    SUPERPOSED candidates.
  const candidates = applyLoopAndCyclePrevention(cell);

  // 3. Single non-zero circuit over all candidates. Guarantees ≥1 OPEN
  //    if k ≥ 1; if k = 0 this is a structural dead end and the player
  //    walks back through the entry wall manually.
  if (candidates.length > 0) {
    await processNonzeroCircuit(cell, candidates);
  }

  // 4. If this is the pre-picked exit cell, open the exit wall deterministically.
  if (cell.r === exitRow && cell.c === GRID - 1) {
    const id = vwallId(exitRow, GRID - 1);
    if (walls[id].state !== 'OPEN') {
      walls[id].state = 'OPEN';
      stats.wallsCollapsed++;
      onExitOpened(exitRow);
    }

  // 5. Orphan check. If the exit is no longer reachable, end the game
  //    honestly — no classical override.
  if (checkExitSealed()) {
    gameOver = true;
    setStatus('★ Quantum entanglement sealed this maze. Restart for a new measurement.');

  inputLocked = false;
}
```

```javascript
  //   R1: at most 1 Bell-paired wall per cell
  //   R2: no walls adjacent to (0,0) or the pre-picked exit cell
  //   R3: Manhattan distance between paired walls ≥ 5
  //   R4: only non-border, non-exit interior walls
  //   R5: target ~10% pair density (i.e. ~20% of interior walls entangled)

  const protectedCells = new Set(['0,0', `${exitRow},${GRID - 1}`]);
  const cellHasBellWall = new Set();  // cell keys whose count of Bell walls is already 1

  const eligible = Object.keys(walls).filter(id => {
    const w = walls[id];
    if (w.isBorder || w.isExit) return false;
    // Walls touching protected cells are excluded.
    const [type, rs, cs] = id.split(':');
    const r = +rs, c = +cs;
    // The two cells this wall sits between:
    let a, b;
    if (type === 'H') {
      a = `${r},${c}`;
      b = `${r + 1},${c}`;
    } else {
      a = `${r},${c}`;
      b = `${r},${c + 1}`;
    }
    if (protectedCells.has(a) || protectedCells.has(b)) return false;
    return true;
  });
  shuffle(eligible);

  const targetPairs = Math.floor(eligible.length * 0.10);
  const used = new Set();
  let made = 0;

  // Helper: returns the two cell-keys flanking a wall.
  function flank(id) {
    const [type, rs, cs] = id.split(':');
    const r = +rs, c = +cs;
    if (type === 'H') return [`${r},${c}`, `${r + 1},${c}`];
    return [`${r},${c}`, `${r},${c + 1}`];
  }

  for (const w of eligible) {
    if (made >= targetPairs) break;
    if (used.has(w)) continue;
    // R1: skip if either flanking cell already has a Bell wall.
    const [wa, wb] = flank(w);
    if (cellHasBellWall.has(wa) || cellHasBellWall.has(wb)) continue;

    let best = null, bestDist = -1;
    for (const cand of eligible) {
      if (cand === w || used.has(cand)) continue;
      const [ca, cb] = flank(cand);
      if (cellHasBellWall.has(ca) || cellHasBellWall.has(cb)) continue;
      const d = wallDist(w, cand);
      if (d > bestDist) { best = cand; bestDist = d; }
    }
    if (best && bestDist >= 5) {
      walls[w].state = 'ENTANGLED';
      walls[w].bellPartner = best;
      walls[best].state = 'ENTANGLED';
      walls[best].bellPartner = w;
      used.add(w); used.add(best);
      const [wa1, wb1] = flank(w);
      const [wa2, wb2] = flank(best);
      cellHasBellWall.add(wa1); cellHasBellWall.add(wb1);
      cellHasBellWall.add(wa2); cellHasBellWall.add(wb2);
      made++;
    }
  }
}
```

```python
Assumes Flask server is running on :5000 and a static server on :8765

import json
import unittest
from playwright.sync_api import sync_playwright


GAME_URL = "http://localhost:8765/index.html"


class TestQuantumMazeV03(unittest.TestCase):
    def setUp(self):
        self.pw = sync_playwright().start()
        self.browser = self.pw.chromium.launch()
        self.page = self.browser.new_page()
        self.page.goto(GAME_URL)
        # Wait for the exit-row pick to complete.
        self.page.wait_for_function("typeof exitRow === 'number'", timeout=10000)

        self.browser.close()
        self.pw.stop()

    def _play_one_game(self):
        """Auto-drive the player until the maze is done or sealed.

        Returns a dict of game-state snapshots."""
        return self.page.evaluate("""
        async () => {
          await start();
          await new Promise(r => setTimeout(r, 80));
          const startExitRow = exitRow;
          let safety = 4000;
          let sealedAt = null;
          while (safety-- > 0) {
            if (gameOver) { sealedAt = { r: player.r, c: player.c }; break; }
            let moved = false;
            for (const side of ['N','E','S','W']) {
              const id = cellWalls(player.r, player.c)[side];
              const w = walls[id];
              if (!w || w.state !== 'OPEN') continue;
              const dr = side === 'N' ? -1 : side === 'S' ? 1 : 0;
              const dc = side === 'E' ? 1 : side === 'W' ? -1 : 0;
              const nr = player.r + dr, nc = player.c + dc;
              if (nr < 0 || nr >= GRID || nc < 0 || nc >= GRID) continue;
              if (cells[nr][nc].visited) continue;
              player.r = nr; player.c = nc;
              await enterCell({ r: nr, c: nc });
              moved = true;
              break;
            }
            if (!moved) break; // no auto-backtrack in 0.3 — done
          }
          // Count visited, OPEN walls, etc.
          let visitedCount = 0;
          for (let r = 0; r < GRID; r++) for (let c = 0; c < GRID; c++)
            if (cells[r][c].visited) visitedCount++;
          let openCount = 0, solidCount = 0, superCount = 0;
          for (const id in walls) {
            if (walls[id].state === 'OPEN') openCount++;
            else if (walls[id].state === 'SOLID') solidCount++;
            else if (walls[id].state === 'SUPERPOSED') superCount++;
          }
          return {
            startExitRow,
            exitOpenedRow,
            visitedCount,
            openCount, solidCount, superCount,
            sealed: gameOver,
            sealedAt,
          };
        }
        """)

    def test_50_games_invariants(self):

        seal_count = 0
        exit_rows_seen = []
        for i in range(50):
            r = self._play_one_game()

            if r["sealed"]:

                continue

            # - exit row was picked uniformly (we'll check distribution after)

            self.assertLess(r["startExitRow"], 15)

            # - no superposed walls reachable from the start (anything

                             f"orphan rate too high: {seal_count}/50 games sealed")
        # Exit row should hit a variety of rows (not always the same).
        # Chi-squared style: at least 5 distinct rows across 50 games.
        self.assertGreaterEqual(len(set(exit_rows_seen)), 5,
                                f"exit-row distribution too narrow: {set(exit_rows_seen)}")
```
